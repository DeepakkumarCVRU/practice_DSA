mkdir my-node-ts-app
cd my-node-ts-app
npm init -y
npm install --save-dev typescript @types/node
npm install --save-dev tsx   # modern alternative to ts-node
npx tsc --init
"type": "module"
mkdir src
# Create src/index.ts AND WRITE CODE 

Add Scripts to package.jsonAdd these under "scripts":
"scripts": {
  "build": "tsc",
  "start": "node dist/index.js",
  "dev": "tsx src/index.ts",
  "watch": "tsc --watch"
}
